  root.querySelectorAll('[data-src]').forEach(box => {
    const p = new Image();
    p.onload = () => { box.classList.add('has-img'); box.innerHTML = ''; p.alt = box.dataset.label; box.appendChild(p); };
    p.src = box.dataset.src;
  });
}

/* ===== 5. ROUTER + BACK BUTTON ===== */
const PARENT = { places:'home', about:'home', hotels:'home', place:'places', pay:'places', hotel:'hotels' };
const stack = [];                       // screens visited, so Back returns to the real previous one
function route(keep) {
  const h = location.hash.slice(1) || 'home', [name, id] = h.split('/');
  if (stack.length > 1 && stack[stack.length - 2] === h) stack.pop();
  else if (stack[stack.length - 1] !== h) stack.push(h);
  $('app').innerHTML = (SCREENS[name] || SCREENS.home)(id);
  $('app').querySelectorAll('[data-back]').forEach(b => b.textContent = t('back'));
  hydrate($('app')); if (!keep) window.scrollTo(0, 0);
  document.querySelector('.lang').hidden = name === 'home';   // home has the open language panel
}
function goBack() {
  if (stack.length > 1) location.hash = stack[stack.length - 2];
  else location.hash = PARENT[(location.hash.slice(1) || 'home').split('/')[0]] || 'home';
}
$('app').addEventListener('click', e => {
  const g = e.target.closest('[data-go]'), b = e.target.closest('[data-back]');
  const l = e.target.closest('[data-lang]'), m = e.target.closest('[data-more]');
  if (l) setLang(l.dataset.lang);                       // language buttons on the home panel
  else if (m) { moreOpen = !moreOpen; route(true); }    // expand / collapse extra languages
  else if (g) location.hash = g.dataset.go; else if (b) goBack();
});
window.addEventListener('hashchange', () => route());

/* ===== 6. LANGUAGE DROPDOWN ===== */
function setLang(code) {
  lang = code;
  document.documentElement.lang = code;
  document.documentElement.dir = ALL[code][1] ? 'rtl' : 'ltr';
  document.querySelectorAll('[data-t]').forEach(el => el.textContent = t(el.dataset.t));
  $('langName').textContent = ALL[code][0];
  $('langMore').textContent = $('langExtra').hidden ? t('more') : t('less');
  document.querySelectorAll('.opt').forEach(b => b.classList.toggle('on', b.dataset.code === code));
  try { localStorage.setItem('qonaqLang', code); } catch (e) {}   // remembered between screens and visits
  route(true);
}
function buildMenu(list, box) {
  Object.keys(list).forEach(c => {
    const b = document.createElement('button');
    b.className = 'opt'; b.dataset.code = c; b.textContent = list[c][0];
    b.onclick = () => { $('langMenu').hidden = true; setLang(c); };
    box.appendChild(b);
  });
}
